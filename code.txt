function showPage(page){
 document.getElementById("home").classList.add("hidden");
 document.getElementById("tutor").classList.add("hidden");
 document.getElementById("summary").classList.add("hidden");

 document.getElementById(page).classList.remove("hidden");
}

function explainTopic(){

 let topic=document.getElementById("topic").value.trim();

 if(topic===""){
  alert("Please enter a topic");
  return;
 }

 document.getElementById("tutorOutput").innerHTML=
 `<h3>${escapeHTML(topic)}</h3>
 <b>1. Introduction</b>
 <p>${escapeHTML(topic)} is the topic you want to understand.
 Start by learning its definition, purpose and important concepts.</p>

 <b>2. Step-by-step learning</b>
 <ol>
 <li>Understand the basic definition.</li>
 <li>Identify the main concepts.</li>
 <li>Study a simple example.</li>
 <li>Learn its real-world applications.</li>
 <li>Revise the important points.</li>
 </ol>

 <b>3. Simple Analogy</b>
 <p>Think of learning this topic like understanding a machine:
 first learn its purpose, then its parts, and finally how they work together.</p>

 <b>4. Check Your Understanding</b>
 <p>Can you explain ${escapeHTML(topic)} in your own words
 and give one example?</p>`;
}

function summarizeNotes(){

 let text=document.getElementById("notes").value.trim();

 if(text.length<40){
  alert("Please enter more study material");
  return;
 }

 let sentences=text.match(/[^.!?]+[.!?]+|[^.!?]+$/g)||[];

 let words=text.toLowerCase().match(/[a-z]{4,}/g)||[];

 let frequency={};

 let stop=["this","that","with","from","have","were",
 "they","them","their","there","then","into","about",
 "which","while","when","what","your","you","are",
 "for","the","and","but","not","can","will","was"];

 words.forEach(word=>{
  if(!stop.includes(word)){
   frequency[word]=(frequency[word]||0)+1;
  }
 });

 let important=Object.keys(frequency)
 .sort((a,b)=>frequency[b]-frequency[a])
 .slice(0,8);

 let scored=sentences.map((sentence,index)=>{
  let count=0;
  sentence.toLowerCase().split(/\W+/).forEach(word=>{
   count+=frequency[word]||0;
  });
  return {sentence,index,score:count};
 });

 let selected=scored
 .sort((a,b)=>b.score-a.score)
 .slice(0,Math.max(2,Math.ceil(sentences.length*.25)))
 .sort((a,b)=>a.index-b.index);

 let summary=selected.map(x=>x.sentence).join(" ");

 let questions=important.slice(0,3).map(word=>
  "What does "+word+" mean in these notes?"
 );

 document.getElementById("summaryOutput").innerHTML=
 `<h3>TL;DR Summary</h3>
 <p>${escapeHTML(summary)}</p>

 <h3>Important Key Terms</h3>
 <ul>${important.map(w=>"<li>"+escapeHTML(w)+"</li>").join("")}</ul>

 <h3>Self-Check Questions</h3>
 <ol>${questions.map(q=>"<li>"+escapeHTML(q)+"</li>").join("")}</ol>`;
}

function escapeHTML(value){
 return value.replace(/[&<>"']/g, char => ({
  '&':'&amp;',
  '<':'&lt;',
  '>':'&gt;',
  '"':'&quot;',
  "'":'&#39;'
 }[char]));
}
